async function handleLogin(username, handler, password) { try { let user = null; let loginIdentifier = username; // Try login with handler first if in login mode if (handler.trim()) { const handlerToTry = handler.startsWith('@') ? handler : '@' + handler; const handlerResponse = await fetch(`${TURSO_API_BASE}/users?handler=${encodeURIComponent(handlerToTry)}`, { method: 'GET', headers: headers }); if (handlerResponse.ok) { user = await handlerResponse.json(); loginIdentifier = handlerToTry; } } // If handler didn't work, try username if (!user) { const response = await fetch(`${TURSO_API_BASE}/users?username=${encodeURIComponent(username)}`, { method: 'GET', headers: headers }); if (response.status === 404) { showMessage('Incorrect credentials.'); return; } if (!response.ok) throw new Error('Verification endpoint offline.'); user = await response.json(); loginIdentifier = username; } // Check if password is missing BEFORE attempting password match if (user && isPasswordMissing(user.password)) { showMessage('Your password was deleted in the great data move. Please set a new one.', false); openResetPasswordModal({ id: user.id, username: user.username || loginIdentifier }); return; } // Only reach here if password is NOT missing if (!user) { showMessage('Incorrect credentials.'); return; } const typedHash = await sha256Hex(password); if (passwordsMatch(user.password, password, typedHash)) { showMessage('Login successful! Redirecting...', false); localStorage.setItem('chatUser', user.username || loginIdentifier); setTimeout(() => { window.location.href = '../index.html'; }, 1500); } else { showMessage('Incorrect credentials.'); } } catch (err) { showMessage('Connection error: ' + err.message); console.error(err); } }
